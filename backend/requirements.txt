# VeriVision Backend Dependencies
#
# FastAPI Cloud (and other non-Docker hosts) install straight from this
# file with no apt/system-package step and no way to run a separate
# "install CPU-only torch first" command like the Dockerfile does. So:
# - torch/torchvision are pinned to their +cpu build explicitly (pip won't
#   pick the +cpu wheel over the much larger default CUDA wheel on PyPI
#   unless the version string forces it) and resolved from the extra
#   index below.
# - opencv-python-headless replaces opencv-python: the full build needs
#   libGL/X11 shared libraries that aren't installable without apt access;
#   headless skips that GUI code path entirely (this API never opens a
#   window) and imports fine on a bare Python environment.
--extra-index-url https://download.pytorch.org/whl/cpu

# [standard] pulls in fastapi-cli, which provides the `fastapi run`
# command FastAPI Cloud's container starts with — plain `fastapi` installs
# the package but not that CLI, so the container crash-loops without it.

uvicorn[standard]==0.51.0
